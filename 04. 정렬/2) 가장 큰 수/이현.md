```js
function solution(numbers) {
  const tmp = numbers
    .toSorted((a, b) => {
      const strA = a.toString();
      const strB = b.toString();
      const AB = Number(strA + strB);
      const BA = Number(strB + strA);
      return BA - AB;
    })
    .join("");

  return tmp.startsWith("0") ? "0" : tmp;
}
```
