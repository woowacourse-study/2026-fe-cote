function solution(citations) {
  citations.sort((a, b) => a - b);
  let canIndex = 0;
  let maxH = 0;
  for (let i = 0; i < citations[citations.length - 1]; i++) {
    if (citations.slice(canIndex).length >= i) {
      maxH = i;
      if (maxH >= citations[canIndex]) canIndex++;
    } else return maxH;
  }

  return 0;
}
