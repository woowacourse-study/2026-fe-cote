```js
function solution(sizes) {
  for (let i = 0; i < sizes.length; i++) {
    const [w, h] = sizes[i];
    if (w < h) {
      const tmp = sizes[i][0];
      sizes[i][0] = sizes[i][1];
      sizes[i][1] = tmp;
    }
  }

  return (
    Math.max(...sizes.map(([w, _]) => w)) *
    Math.max(...sizes.map(([_, h]) => h))
  );
}

// 2개중 큰거를 한쪽으로 다 몰아
// 그리고 양쪽에서 max를 찾으면 됨
```
